function renderProviderOptions() { // حفظ الاختيارات اليدوية الحالية قبل إعادة بناء القائمة const previousSelected = new Set(); if (providerSelectionMode === "manual") { PROVIDERS.forEach(name => { const checkbox = document.getElementById(`provider-${name}`); if (checkbox && checkbox.checked) { previousSelected.add(name); } }); } providerOptions.innerHTML = ""; // ========================= // Automatic // ========================= const automaticOption = document.createElement("label"); automaticOption.className = "provider-option"; automaticOption.innerHTML = ` 🤖 تلقائي `; providerOptions.appendChild(automaticOption); const automaticRadio = automaticOption.querySelector("input"); automaticRadio.addEventListener("change", () => { if (automaticRadio.checked) { providerSelectionMode = "automatic"; updateProviderSelectionInfo(); } }); // ========================= // المنصات المتاحة فقط // ========================= const availableProviders = PROVIDERS.filter(name => { return String(providerStates[name] || "unknown").toLowerCase() === "available"; }); availableProviders.forEach(name => { const option = document.createElement("label"); option.className = "provider-option"; const labelName = { deepseek: "DeepSeek", gemini: "Gemini", mistral: "Mistral" }[name] || name; option.innerHTML = ` ${labelName} `; providerOptions.appendChild(option); const checkbox = option.querySelector("input"); // استرجاع الاختيار السابق إذا كانت المنصة ما زالت متاحة if ( providerSelectionMode === "manual" && previousSelected.has(name) ) { checkbox.checked = true; } checkbox.addEventListener("change", () => { providerSelectionMode = "manual"; // إلغاء Automatic عند اختيار منصة يدويًا automaticRadio.checked = false; updateProviderSelectionInfo(); }); }); // إذا لم توجد أي منصة متاحة للاختيار اليدوي if (availableProviders.length === 0) { const emptyMessage = document.createElement("div"); emptyMessage.className = "provider-selection-info"; emptyMessage.style.gridColumn = "1 / -1"; emptyMessage.textContent = "لا توجد منصة متاحة حاليًا للاختيار اليدوي. يمكنك استخدام «تلقائي»."; providerOptions.appendChild(emptyMessage); } updateProviderSelectionInfo(); }
